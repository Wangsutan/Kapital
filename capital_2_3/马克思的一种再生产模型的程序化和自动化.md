def print_categories(category_1, category_2):
    print("Ⅰ:", category_1)
    print("Ⅱ:", category_2)


def print_split_line(word):
    print("-" * 6, word, "-" * 6)


def produce(category, surplus_value_rate):
    category["m"] = category["v"] * surplus_value_rate


def calc_total_value(category_1, category_2):
    print_split_line("calc total value")
    for category in [category_1, category_2]:
        print("calculating:", category)
        total_ctg = 0
        for x in category.values():
            total_ctg += x
        print("total_ctg: %.2f" % total_ctg)

    total_value = 0
    for category in [category_1, category_2]:
        # print("calculating:", category)
        for x in category.values():
            total_value += x
    print("total value: %.2f" % total_value)


def calc_cvm(category_1, category_2):
    print_split_line("calc cvm")
    cvm = list(category_1)
    for item in cvm:
        total_item = category_1[item] + category_2[item]
        print("total %s: %.2f" % (item, total_item))


def accumulate(category_1, category_2):
    print_split_line("accumulating")
    acc_m_1 = category_1["m"] / 2
    print("acc_m_1:", acc_m_1)

    add_to_c_1 = acc_m_1 * add_c_rate_1
    print("add_to_c_1:", add_to_c_1)

    add_to_v_1 = acc_m_1 * add_v_rate_1
    print("add_to_v_1:", add_to_v_1)

    category_1["c"] += add_to_c_1
    category_1["v"] += add_to_v_1
    category_1["m"] -= acc_m_1

    add_to_c_2 = category_1["v"] + \
        category_1["m"] - category_2["c"]    # old c_2
    print("add_to_c_2:", add_to_c_2)
    category_2["c"] += add_to_c_2

    add_to_v_2 = category_2["c"] * v_c_rate_2 - category_2["v"]    # old v_2
    print("add_to_v_2:", add_to_v_2)
    category_2["v"] += add_to_v_2

    category_2["m"] -= (add_to_c_2 + add_to_v_2)

    print_split_line("expanded")
    print_categories(category_1, category_2)

    return category_1, category_2


print("Expanded reproduction of Marx.")

add_v_rate_1 = category_1["v"] / (category_1["c"] + category_1["v"])
v_c_rate_2 = category_2["v"] / (category_2["c"])

for i in range(1, 7):
    print(f"\nNum.{i}")

    produce(category_1, 1.00)
    produce(category_2, 1.00)

    print_split_line("produce")
    print_categories(category_1, category_2)

    calc_total_value(category_1, category_2)
    calc_cvm(category_1, category_2)

    category_1, category_2 = accumulate(category_1, category_2)

    category_1["m"] = 0
